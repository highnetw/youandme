## 📋 Chapter 1: 인증 및 권한 관리 (Authentication & Authorization)

### 현재 설계 검토
Supabase Auth 선택은 적절합니다. 하지만 몇 가지 보완이 필요합니다.

### 개선 및 추가 사항

**1-1. 사용자 프로필 테이블 구조 개선**
```sql
-- profiles 테이블
CREATE TABLE profiles (
  id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  email TEXT NOT NULL,
  name TEXT,
  phone TEXT,
  role TEXT DEFAULT 'user' CHECK (role IN ('user', 'admin')),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 자동 프로필 생성 트리거
CREATE OR REPLACE FUNCTION handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO public.profiles (id, email, name)
  VALUES (NEW.id, NEW.email, NEW.raw_user_meta_data->>'name');
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
```

**1-2. Row Level Security (RLS) 정책 필수**
```sql
-- 사용자는 본인 프로필만 읽기/수정 가능
ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Users can view own profile"
  ON profiles FOR SELECT
  USING (auth.uid() = id);

CREATE POLICY "Users can update own profile"
  ON profiles FOR UPDATE
  USING (auth.uid() = id);
```

**1-3. 권한 관리 전략**
- `role` 컬럼: 'user', 'admin' 외에 'manager'(예약 관리만), 'content_editor'(콘텐츠 관리만) 추가 고려
- **토큰 갱신 전략**: Supabase는 기본 1시간 후 토큰 만료. `refreshSession()` 자동 호출 로직 구현 필요
- **소셜 로그인 우선순위**: 카카오 > 네이버 > 구글 (국내 사용자 기준)

**1-4. Middleware 구현 예시**
```typescript
// middleware.ts
import { createMiddlewareClient } from '@supabase/auth-helpers-nextjs'
import { NextResponse } from 'next/server'
import type { NextRequest } from 'next/server'

export async function middleware(req: NextRequest) {
  const res = NextResponse.next()
  const supabase = createMiddlewareClient({ req, res })
  
  const { data: { session } } = await supabase.auth.getSession()
  
  // 보호된 라우트 체크
  if (req.nextUrl.pathname.startsWith('/admin') && session?.user?.user_metadata?.role !== 'admin') {
    return NextResponse.redirect(new URL('/', req.url))
  }
  
  return res
}
```

```typescript
// 체크아웃 후 14일 이내, 실제 숙박한 사람만 작성 가능
async function canWriteReview(userId: string, bookingId: string) {
  const { data: booking } = await supabase
    .from('bookings')
    .select('user_id, check_out, status')
    .eq('id', bookingId)
    .single();
    
  if (!booking || booking.user_id !== userId) return false;
  if (booking.status !== 'completed') return false;
  
  const daysSinceCheckout = (Date.now() - new Date(booking.check_out).getTime()) / (1000 * 60 * 60 * 24);
  if (daysSinceCheckout > 14) return false;
  
  // 이미 작성한 후기가 있는지 확인
  const { data: existingReview } = await supabase
    .from('reviews')
    .select('id')
    .eq('booking_id', bookingId)
    .single();
    
  return !existingReview;
}
```

```sql
-- rooms 테이블에 캐싱 컬럼 추가
ALTER TABLE rooms ADD COLUMN avg_rating DECIMAL(2,1);
ALTER TABLE rooms ADD COLUMN review_count INTEGER DEFAULT 0;

-- 트리거로 자동 업데이트
CREATE OR REPLACE FUNCTION update_room_rating()
RETURNS TRIGGER AS $$
BEGIN
  UPDATE rooms
  SET 
    avg_rating = (SELECT AVG(rating) FROM reviews WHERE room_id = NEW.room_id AND is_verified = TRUE),
    review_count = (SELECT COUNT(*) FROM reviews WHERE room_id = NEW.room_id AND is_verified = TRUE)
  WHERE id = NEW.room_id;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_update_room_rating
AFTER INSERT OR UPDATE ON reviews
FOR EACH ROW EXECUTE FUNCTION update_room_rating();
```

```sql
-- 좋아요 기능 추가
CREATE TABLE review_likes (
  user_id UUID REFERENCES profiles(id) ON DELETE CASCADE,
  review_id UUID REFERENCES reviews(id) ON DELETE CASCADE,
  PRIMARY KEY (user_id, review_id)

-- 베스트 리뷰 쿼리 (평점 + 좋아요 + 최신순)
SELECT r.*, COUNT(rl.user_id) as like_count
FROM reviews r
LEFT JOIN review_likes rl ON r.id = rl.review_id
WHERE r.is_verified = TRUE
GROUP BY r.id
ORDER BY (r.rating * 10 + COUNT(rl.user_id)) DESC, r.created_at DESC
LIMIT 3;
```

```sql
-- PostgreSQL Function (동시성 제어)
CREATE OR REPLACE FUNCTION create_booking_safe(
  p_room_id UUID,
  p_user_id UUID,
  p_check_in DATE,
  p_check_out DATE,
  p_total_price INTEGER
)
RETURNS UUID AS $$
DECLARE
  v_booking_id UUID;
  v_conflict_count INTEGER;
BEGIN
  -- 해당 방을 행 잠금 (다른 트랜잭션 대기)
  PERFORM * FROM rooms WHERE id = p_room_id FOR UPDATE;
  
  -- 중복 예약 체크
  SELECT COUNT(*) INTO v_conflict_count
  FROM bookings
  WHERE room_id = p_room_id
    AND status IN ('confirmed', 'pending')
    AND NOT (p_check_out <= check_in OR p_check_in >= check_out);
  
  IF v_conflict_count > 0 THEN
    RAISE EXCEPTION '이미 예약된 날짜입니다';
  END IF;
  
  -- 예약 생성
  INSERT INTO bookings (room_id, user_id, check_in, check_out, total_price)
  VALUES (p_room_id, p_user_id, p_check_in, p_check_out, p_total_price)
  RETURNING id INTO v_booking_id;
  
  RETURN v_booking_id;
END;
$$ LANGUAGE plpgsql;
```

**4-3. 예약 가능 날짜 조회 (캘린더)**
```typescript
// 특정 월의 예약 불가 날짜 반환
async function getUnavailableDates(roomId: string, year: number, month: number) {
  const startDate = `${year}-${month.toString().padStart(2, '0')}-01`;
  const endDate = new Date(year, month, 0).toISOString().split('T')[0];
  
  const { data } = await supabase
    .from('bookings')
    .select('check_in, check_out')
    .eq('room_id', roomId)
    .in('status', ['confirmed', 'pending'])
    .or(`check_in.lte.${endDate},check_out.gte.${startDate}`);
    
  // 날짜 범위를 배열로 변환
  const unavailableDates = new Set<string>();
  data?.forEach(booking => {
    let currentDate = new Date(booking.check_in);
    const lastDate = new Date(booking.check_out);
    
    while (currentDate < lastDate) {
      unavailableDates.add(currentDate.toISOString().split('T')[0]);
      currentDate.setDate(currentDate.getDate() + 1);
    }
  });
  
  return Array.from(unavailableDates);
}
```

**4-5. 취소 정책 자동화**
```sql
-- 취소 수수료 계산 함수
CREATE OR REPLACE FUNCTION calculate_refund_amount(p_booking_id UUID)
RETURNS INTEGER AS $$
DECLARE
  v_total_price INTEGER;
  v_check_in DATE;
  v_days_until_checkin INTEGER;
  v_refund_rate DECIMAL;
BEGIN
  SELECT total_price, check_in INTO v_total_price, v_check_in
  FROM bookings WHERE id = p_booking_id;
  
  v_days_until_checkin := v_check_in - CURRENT_DATE;
  
  -- 취소 정책 (예: 7일 전 100%, 3일 전 50%, 당일 0%)
  IF v_days_until_checkin >= 7 THEN
    v_refund_rate := 1.0;
  ELSIF v_days_until_checkin >= 3 THEN
    v_refund_rate := 0.5;
  ELSE
    v_refund_rate := 0.0;
  END IF;
  
  RETURN FLOOR(v_total_price * v_refund_rate);
END;
$$ LANGUAGE plpgsql;
```

```sql
-- PostGIS 대신 earthdistance 모듈 사용
CREATE EXTENSION IF NOT EXISTS cube;
CREATE EXTENSION IF NOT EXISTS earthdistance;

-- 펜션으로부터 5km 이내 명소 검색
SELECT 
  s.*,
  earth_distance(
    ll_to_earth(s.latitude, s.longitude),
    ll_to_earth(37.5665, 126.9780) -- 펜션 좌표
  ) AS distance_meters
FROM spots s
WHERE earth_box(ll_to_earth(37.5665, 126.9780), 5000) @> ll_to_earth(s.latitude, s.longitude)
ORDER BY distance_meters;
```

```typescript
    // 명소 마커 (카테고리별 색상)
    spots.forEach(spot => {
      const marker = new kakao.maps.Marker({
        position: new kakao.maps.LatLng(spot.latitude, spot.longitude),
        title: spot.name
      });
      
      // 클릭 시 정보창
      kakao.maps.event.addListener(marker, 'click', () => {
        const infowindow = new kakao.maps.InfoWindow({
          content: `
            <div style="padding:10px;">
              <strong>${spot.name}</strong><br/>
              <span>${spot.category_name}</span><br/>
              <a href="/spots/${spot.id}">상세보기</a>
            </div>
          `
        });
        infowindow.open(map, marker);
      });
      
      marker.setMap(map);
    });
    
  }, [spots]);
  
  return <div ref={mapRef} style={{ width: '100%', height: '500px' }} />;
}
```

```sql
-- spots 테이블에 방문자 수 캐싱
ALTER TABLE spots ADD COLUMN visit_count INTEGER DEFAULT 0;

CREATE OR REPLACE FUNCTION update_spot_visit_count()
RETURNS TRIGGER AS $$
BEGIN
  UPDATE spots
  SET visit_count = (SELECT COUNT(DISTINCT user_id) FROM spot_visits WHERE spot_id = NEW.spot_id)
  WHERE id = NEW.spot_id;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;
```

```typescript
// middleware.ts에 추가
import { Ratelimit } from "@upstash/ratelimit";
import { Redis } from "@upstash/redis";

const ratelimit = new Ratelimit({
  redis: Redis.fromEnv(),
  limiter: Ratelimit.slidingWindow(10, "1 m"), // 분당 10회
});

export async function middleware(request: NextRequest) {
  if (request.nextUrl.pathname.startsWith('/api/')) {
    const ip = request.ip ?? '127.0.0.1';
    const { success } = await ratelimit.limit(ip);
    
    if (!success) {
      return new NextResponse('Too Many Requests', { status: 429 });
    }
  }
}
```
